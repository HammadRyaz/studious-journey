
import java.util.List;
import java.util.Random;

public class AIPlayer extends Player {
    private String difficulty;
    private Random random;

    public AIPlayer(String name, String symbol, String difficulty) {
        super(name, symbol);
        this.difficulty = difficulty;
        this.random = new Random();
    }

    @Override
    public int[] makeMove(GameBoard board) {
        if ("Hard".equals(difficulty)) {
            return hardMove(board);
        }
        if ("Medium".equals(difficulty)) {
            return mediumMove(board);
        }
        return randomMove(board);
    }

    private int[] randomMove(GameBoard board) {
        List<int[]> empty = board.getEmptyCells();
        if (empty.isEmpty()) {
            return null;
        }
        return empty.get(random.nextInt(empty.size()));
    }

    private int[] mediumMove(GameBoard board) {
        int[] win = findWinningMove(board, getSymbol());
        if (win != null) {
            return win;
        }
        return randomMove(board);
    }

    private int[] hardMove(GameBoard board) {
        int[] win = findWinningMove(board, getSymbol());
        if (win != null) {
            return win;
        }
        String opponent = getSymbol().equals("X") ? "O" : "X";
        int[] block = findWinningMove(board, opponent);
        if (block != null) {
            return block;
        }
        if (board.getCell(1, 1).isEmpty()) {
            return new int[] { 1, 1 };
        }
        return randomMove(board);
    }

    private int[] findWinningMove(GameBoard board, String symbol) {
        List<int[]> empty = board.getEmptyCells();
        for (int[] pos : empty) {
            int row = pos[0];
            int col = pos[1];
            board.setCell(row, col, symbol);
            boolean canWin = !board.checkWinner().isEmpty();
            board.clearCell(row, col);
            if (canWin) {
                return pos;
            }
        }
        return null;
    }
}
